\chapter{Algorithm Pseudocode}
\label{app:pseudocode}

Compact pseudocode for the load-bearing algorithms, matching the
production sources line for line in structure. Notation:
$y = (t, r, \theta, \phi, p_r, p_\theta)$ is the Mino-time state,
$c = (M, a, E, L_z, Q)$ the ray constants.

\section{Adaptive step with constraint projection}
(\texttt{RaytraceCpu::trace} inner loop;
\texttt{DormandPrince54::try\_step})

\begin{verbatim}
h = 1e-3
while steps < max_steps:
    (accepted, err, h_next) = dopri5_step(rhs, tau, y, h)   # 7 stages
    h = clamp(h_next, 1e-14, h_max)
    if not accepted: continue
    # constraint projection (skip inside turning-point guard band)
    if R(y.r) >= 0:      y.p_r     = sign(y.p_r)     * sqrt(R(y.r))
    if Theta(y.th) >= 0: y.p_theta = sign(y.p_theta) * sqrt(Theta(y.th))
    if y.r < 1.001*r_horizon + eps:  return CAPTURED
    if y.r > r_escape:               return ESCAPED
    handle_equator_crossing(y_prev, y)          # Section B.3
\end{verbatim}

\section{Grid-free boundary bisection}
(\texttt{shadow\_boundary\_angle})

\begin{verbatim}
lo, hi = 0, angle_max            # capture at lo, escape at hi
repeat 60 times or until hi - lo < 1e-12:
    mid = (lo + hi) / 2
    outcome = trace(ray_at_camera_angle(mid, psi))
    if outcome == CAPTURED: lo = mid else: hi = mid
return (lo + hi) / 2
\end{verbatim}

\section{Equator-crossing localisation and sub-ring count}
(\texttt{trace} disk logic)

\begin{verbatim}
if cos(theta_prev) * cos(theta_now) < 0:        # plane crossed
    bisect substep 40 times on cos(theta) -> crossing state x
    if disk_enabled and r_in <= x.r <= r_out:
        g = 1 / (u_t_circ(x.r) * (E - Omega(x.r) * L_z))
        return DISK_HIT(r=x.r, phi=x.phi, n=crossings, g=g)
    crossings += 1
\end{verbatim}

\section{Least-squares fit and explicit covariance}
(\texttt{OrbitFitter::fit}, \texttt{::covariance})

\begin{verbatim}
theta = start
theta = ceres_LM_minimise(residuals(theta))     # numeric-diff Jacobians
# covariance: explicit central differences, physically scaled steps
for k in 1..13:
    J[k] = (residuals(theta + s_k e_k) - residuals(theta - s_k e_k))
           / (2 s_k)
C = gauss_jordan_inverse(J^T J)                 # NO SVD truncation
sigma_k = sqrt(C[k][k])
\end{verbatim}

\section{Ensemble MCMC over the chi-squared server}
(\texttt{scripts/run\_mcmc.py})

\begin{verbatim}
spawn K persistent workers: fit_s2_tool --chi2 <data> <mode>
log_prob(batch X):
    out-of-bounds rows -> -inf
    distribute rows round-robin to workers; read chi2 back in order
    return -chi2 / 2
walkers = optimum + 0.5 * sigma * N(0,1)        # 32 walkers
run emcee stretch moves for 6000 steps (vectorised log_prob)
tau = max autocorrelation time; discard 4*tau; thin tau/2
save chain + 16/50/84 percentiles
\end{verbatim}

\section{Darwin turning-point initialisation}
(\texttt{S2Model} setup; Section~\ref{sec:darwin})

\begin{verbatim}
a_phys = a_mas * R0                       # angular -> physical (au -> M)
r_p, r_a = a_phys*(1-e), a_phys*(1+e)
# solve R(r_p) = R(r_a) = 0 for E^2, L^2 in the well-conditioned form
#   E^2 = (1 - 2M/r)(1 + L^2/r^2)  at both radii  -> L^2 first
L2 = (f_p/r_p^2 - f_a/r_a^2)^-1 * (f_a - f_p)     # f = 1 - 2M/r
E2 = f_p * (1 + L2/r_p^2)
u  = (u^t from normalisation quadratic, future root)   # solve_ut
state = (t=0, r=r_p, phi=0, u^r=0, u^phi=L/r_p^2, u^t)
integrate one full radial period; cache adaptive nodes (pos + vel)
\end{verbatim}

\section{R{\o}mer fixed-point solve per epoch}
(\texttt{SkyProjection}; Section~\ref{sec:romer})

\begin{verbatim}
t_em = t_obs
repeat 3 times:                     # contraction rate |beta_z| < 0.03
    z    = line_of_sight(sample_at(t_em))
    t_em = t_obs - z / c
return sample_at(t_em)              # observables at emission time
\end{verbatim}

\section{Cluster joint fit assembly}
(\texttt{fit\_cluster}; Chapter~\ref{ch:cluster})

\begin{verbatim}
global = (M, R0, x0, y0, vx, vy, vz)           # one block, Plewa priors
for star k in {S2, S1, S8, S12, S13}:
    elements[k] = per-star block (a, e, i, Omega, omega, t_peri)
    refine elements[k] alone (global frozen at start values)
    probe (Omega, omega) +/- 180 deg; keep lowest starting chi^2
build one Ceres problem:
    residuals_k(global, elements[k]) for every epoch of every star
solve; covariance: explicit central-difference J over the full
concatenated 37-vector, gauss_jordan_inverse(J^T J)   # no SVD
subsets: repeat solve for {S2}, {S2,S1}, ... -> convergence ladder
\end{verbatim}

\section{Sub-ring decomposition and width scan}
(\texttt{SubRingExtractor::analyze}, \texttt{scan\_widths})

\begin{verbatim}
render image with per-pixel record (n = crossing count, bolometric I)
for n in {0, 1, 2}:
    F_n = sum of I over pixels with crossing count n
ratios = F_0/F_1, F_1/F_2
for psi in {0, 90, 180, 270} deg:          # image directions
    walk outward along psi from image centre
    band = contiguous run of n==1 pixels   # may be occluded: record none
    width[psi] = outer_edge - inner_edge   # in camera angle
\end{verbatim}
